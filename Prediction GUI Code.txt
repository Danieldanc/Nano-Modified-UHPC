# ============================================================================
#  Model training
# ============================================================================
def train_model():
    arr  = np.array(DATA, dtype=float)
    X, y = arr[:, :7], arr[:, 7]
    Xtr, Xte, ytr, yte = train_test_split(X, y, test_size=0.20, random_state=42)

    # Split model for metric verification
    m_split = DecisionTreeRegressor(**DT_PARAMS)
    m_split.fit(Xtr, ytr)

    # Full-data deployment model for best predictions
    m_deploy = DecisionTreeRegressor(**DT_PARAMS)
    m_deploy.fit(X, y)
    return m_deploy


def strength_class(fc):
    """Strength classification calibrated to 30.4 – 126.8 MPa dataset range."""
    if   fc < 45:   return "Very Low  (< 45 MPa)",  "#b03020"
    elif fc < 65:   return "Low  (45 – 65 MPa)",    "#d4650a"
    elif fc < 85:   return "Moderate  (65 – 85 MPa)","#b89200"
    elif fc < 105:  return "High  (85 – 105 MPa)",   GREEN
    else:           return "Very High  (> 105 MPa)",  BLUE

    # ── actions ───────────────────────────────────────────────────────────────
    def calculate(self):
        try:
            x = [float(self.entries[name].get())
                 for (_, name, _, _) in FEATURES]
        except ValueError:
            messagebox.showerror("Invalid Input",
                                  "Please enter a valid number in every field.")
            return

        fc = float(self.model.predict(np.array([x]))[0])
        fc = max(0.0, fc)

        self.pred_var.set(f"{fc:.2f} MPa")
        label, color = strength_class(fc)
        self.pred_lbl.config(fg=color)
        self.class_var.set(label)
        self.class_lbl.config(fg=color)

    def reset(self):
        for (_, name, _, _) in FEATURES:
            self.entries[name].delete(0, "end")
        self.pred_var.set("—")
        self.pred_lbl.config(fg=ORANGE)
        self.class_var.set("—")
        self.class_lbl.config(fg=GREEN)

    def load_example(self):
        """Load a representative high-strength nano-silica mix."""
        for (_, name, _, default) in FEATURES:
            self.entries[name].delete(0, "end")
            self.entries[name].insert(0, str(default))
        self.calculate()
